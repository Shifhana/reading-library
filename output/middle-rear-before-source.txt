import { books } from '../data/books.ts'

type Point = readonly [number, number]
// Fixed shelf positions: library index, centre, height, lean, seating offset.
// All library records appear once; middle and remaining rear poses stay fixed.
const arrangements = [
  { name: 'rear', bottom: 426.92, slope: .065, scale: 1, slots: [
    [0, 938, 53, -1, 0], [1, 982, 58, .7, -.6], [2, 1027, 48, -1.2, 0],
    [3, 1072, 54, .5, -.5], [4, 1120, 59, 1.3, 0],
  ] },
  { name: 'middle', bottom: 558.51, slope: -.035, scale: 1, slots: [
    [5, 316, 60, -1.1, 0], [6, 373, 72, .6, -.6], [7, 454, 63, -.7, 0],
    [8, 512, 69, 1.2, -.8], [9, 599, 74, -.5, 0], [10, 681, 62, 1.4, -.5],
    [11, 738, 69, -.8, 0], [12, 815, 64, .4, -.7], [13, 874, 68, -1.4, 0],
    [14, 968, 71, .8, -.5],
  ] },
  { name: 'foreground', bottom: 762.23, slope: 0, scale: .935, slots: [
    [15, 1098, 110, -.2, 0], [16, 1196, 117, .15, -.4],
    [17, 1308, 106, -.1, 0], [18, 1410, 115, .25, -.6],
    [19, 1524, 121, -.15, 0], [20, 1628, 114, .1, -.5],
  ] },
] as const

// Approved cover proportions also determine the held-book size. Preserve each
// book's aspect ratio while varying its shelf pose so the selected state stays
// unchanged. These references describe presentation, not new library data.
const referenceWidths = [35, 34, 37, 35, 39, 47, 49, 50, 47, 51, 44, 49, 47, 50, 52, 76, 78, 81, 77, 89, 86]
const referenceHeights = [53, 55, 51, 54, 56, 65, 70, 63, 69, 67, 66, 69, 64, 68, 71, 110, 117, 106, 115, 121, 114]
const referenceLeans = [-1, .7, -1.2, .5, 1.3, -1.1, .6, -.7, 1.2, -.5, 1.4, -.8, .4, -1.4, .8, -.8, 1.1, -.6, .7, -1.2, .6]

// Colours stay tied to library identity, including the transferred books.
const colours = ['#917867', '#d3beaf', '#5a594b', '#845759', '#ac9585', '#4b595d', '#e3d2c7', '#a06f76', '#a1897a', '#6b6a5b', '#decbbf', '#445054', '#455057', '#decbc1', '#9e8677', '#616254', '#926162', '#746c5e', '#c0b29d', '#58666a', '#a79482']
export const displayBooks = arrangements.flatMap(shelf => shelf.slots.flatMap(([index, center, originalHeight, lean, offset]) => {
  const record = books[index]
  const colour = colours[index]
  const thickness = [.038, .046, .031, .042, .035, .051, .033][index % 7]
  if (!record) return []
  const height = originalHeight * shelf.scale
  const referenceSlope = index < 5 ? .065 : index < 15 ? -.035 : .105
  const aspect = referenceWidths[index] / referenceHeights[index] + Math.abs(referenceSlope + Math.tan(referenceLeans[index] * Math.PI / 180))
  const shear = shelf.slope + Math.tan(lean * Math.PI / 180)
  const width = height * (aspect - Math.abs(shear))
  const bottom = shelf.bottom + offset
  const shift = height * shear
  const corners: readonly Point[] = [
    [center - width / 2 + shift, bottom - height], [center + width / 2 + shift, bottom - height],
    [center + width / 2, bottom], [center - width / 2, bottom],
  ]
  const left = Math.min(...corners.map(([x]) => x))
  return [{
    id: `room-book-${record.slug}`, slug: record.slug, record, shelf: shelf.name,
    corners, colour, thickness, shelfThickness: thickness * (shelf.name === 'rear' ? .85 : 1), bottom, top: bottom - height, left,
    width: Math.max(...corners.map(([x]) => x)) - left, height, center,
    label: `${record.title} on the ${shelf.name} shelf`,
    returnLabel: `Return ${record.title} to the ${shelf.name} shelf`,
    outline: corners.map(p => p.join(',')).join(' '),
  }]
}))
